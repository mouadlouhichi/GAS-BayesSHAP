\begin{abstract}
\noindent
Shapley-value estimation for black-box models faces an exponential
combinatorial barrier and, in most practical estimators, an absence of
distribution-free, anytime uncertainty quantification.  We propose
GAS-BayesSHAP, which couples a bounded-linear Bayesian control variate with a
Neyman-stratified residual certifier to deliver point estimates that match
exact Shapley values to $10^{-3}$ RMSE at roughly $70\%$ of the exact
enumeration budget, together with anytime empirical-Bernstein confidence
sequences whose simultaneous coverage we validate at $R=500$ trials.  On real
UCI wine and Beijing multi-site air-quality data ($M=11$), sign-certification
is achievable in the regime where the certified width falls below the
attribution magnitude; we characterize this certification cost frontier and
show how empirical residual-range tightening shrinks the required budget by an
order of magnitude. \emph{(Complete after width-tightening experiments.)}
\end{abstract}

\section{Method}\label{sec:method}
\subsection{Bounded linear control variate}\label{sec:bcv}
The surrogate is $m_b(S)=c+\lambda h(S)$ with
$h(S)=\mathbf{k}_{\mathcal{D}}(S)^\top\boldsymbol{\alpha}$,
$\boldsymbol{\alpha}=(\mathbf{K}_{\mathcal{D}\mathcal{D}}+\eta^2 I)^{-1}\mathbf{y}$,
conservative bounds $h_{\mathrm{lb}},h_{\mathrm{ub}}$ from the kernel range
$[\sigma_0^2\rho^M,\sigma_0^2]$, shrinkage
$\lambda=\min(1,(U-L)/(h_{\mathrm{ub}}-h_{\mathrm{lb}}))$, and shift
$c=L-\lambda h_{\mathrm{lb}}$, guaranteeing $m_b(S)\in[L,U]$ without clipping.
The surrogate attribution is closed-form
$\boldsymbol{\phi}(m_b)=\lambda\mathbf{K}_{\phi,\mathcal{D}}\boldsymbol{\alpha}$
(Lemma~D, $O(M^2)$), with prior covariance $\mathbf{K}_{\phi,\phi}$ (Lemma~E,
$O(M^3)$) and $\lambda^2$-scaled posterior covariance.

\subsection{Neyman-stratified residual certification}\label{sec:cert}
Residuals $R_i(S)=\Delta_i^v(S)-\Delta_i^{m_b}(S)$ are sampled with
add-one/remove-one marginals (Lemma~F).  Extreme singleton strata
$s\in\{0,M-1\}$ are exact (Lemma~G).  Interior strata are allocated by the
coupled adjacent-stratum Neyman program (Theorem~A) and certified by an
anytime empirical-Bernstein sequence (Theorem~B) with per-cell widths
\begin{equation}
W_i^{\mathrm{res}}(\mathbf{n}_i)=\frac1M\sum_{s=1}^{M-2}
\left[\sqrt{\frac{2(\hat\sigma_{i,s}^r)^2\log(\pi^2 M^2 n_{i,s}^2/3\delta)}{n_{i,s}}}
+\frac{7 R_\Delta^{\mathrm{res}}\log(\pi^2 M^2 n_{i,s}^2/3\delta)}{3(n_{i,s}-1)}\right].
\end{equation}

\subsection{Efficiency projection and sign certification}\label{sec:proj}
The raw estimator $\hat{\boldsymbol{\phi}}^{\mathrm{raw}}=
\boldsymbol{\phi}(m_b)+\hat{\boldsymbol{\phi}}(r_{\mathcal{D}})$ is projected
onto $\sum_i\phi_i=\Delta_{\mathrm{total}}$ with posterior-diagonal weights
(Theorem~C); Corollary~C.1 gives post-projection widths
$W_i^{\mathrm{proj}}=W_i^{\mathrm{res}}+(v_i/\sum_j v_j)\sum_j W_j^{\mathrm{res}}$,
and a feature is \emph{sign-certified} iff $|\phi_i^*|>W_i^{\mathrm{proj}}$.
